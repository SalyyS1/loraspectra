Native CLI Quickstart (Zero Dependencies)

      

      


cargo install --git https://github.com/SalyyS1/loraspectra

# Inspect or audit remote HuggingFace LoRAs directly (auto-cached in ${TMPDIR:-/tmp}/loraspectra-cache/)
loraspectra info hf://XLabs-AI/flux-RealismLora
loraspectra audit hf://org/llama3-lora/adapter_model.safetensors

# Compress a rank-collapsed LoRA at 99% Frobenius energy and emit PEFT rank_pattern config
loraspectra compress adapter_model.safetensors adapter_slim.safetensors --energy 0.99 --peft-config adapter_config.json

# Analyze N-adapter subspace collision and norm inflation prior to TIES/DARE/linear merging
loraspectra merge-check hf://org/code-lora hf://org/math-lora hf://org/sql-lora --sarif merge.sarif
